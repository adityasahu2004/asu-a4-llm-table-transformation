Operator plan: Parse the Date column into a datetime object. Format the date as MM/DD/YY for `dt_strata`. Derive the ISO day of week (1=Monday, ..., 7=Sunday) for `dow`. Rename the 24 hourly columns (1:00 through 24:00:00) to the target schema names (`pct_hourly_0100` through `pct_hourly_2400`). Add a `pct_hourly_2500` column filled with NaNs.

```python
import pandas as pd
import numpy as np

def transform(source: pd.DataFrame) -> pd.DataFrame:
    # Parse dates
    dt = pd.to_datetime(source['Date'], format='%a %m/%d/%Y')
    
    # Create output dataframe
    out = pd.DataFrame(index=source.index)
    
    # Format date string
    out['dt_strata'] = dt.dt.strftime('%m/%d/%y')
    
    # Derive day of week (ISO: Mon=1, Sun=7)
    out['dow'] = dt.dt.dayofweek + 1
    
    # Map hourly columns
    for h in range(1, 25):
        src_col = f'{h}:00' if h < 24 else '24:00:00'
        tgt_col = f'pct_hourly_{h:02d}00'
        out[tgt_col] = source[src_col]
        
    # Add 25th hour as NaN
    out['pct_hourly_2500'] = np.nan
    
    return out.reset_index(drop=True)
```
