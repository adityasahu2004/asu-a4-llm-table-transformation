Operator plan: Parse DT_STRATA into datetime; format as "DayOfWeek MM/DD/YYYY" for column `cst`; rename the 24 hourly percentage columns to `1:00` through `24:00`; drop all other columns.

```python
import pandas as pd

def transform(source: pd.DataFrame) -> pd.DataFrame:
    # Parse the date string
    dt = pd.to_datetime(source['DT_STRATA'], format='%m/%d/%Y')
    
    # Create the output dataframe
    out = pd.DataFrame(index=source.index)
    
    # Format cst: "Fri 01/01/2016"
    out['cst'] = dt.dt.strftime('%a %m/%d/%Y')
    
    # Map PCT_HOURLY_XX00 to H:00
    for h in range(1, 25):
        src_col = f'PCT_HOURLY_{h:02d}00'
        tgt_col = f'{h}:00'
        out[tgt_col] = source[src_col]
        
    return out.reset_index(drop=True)
```
