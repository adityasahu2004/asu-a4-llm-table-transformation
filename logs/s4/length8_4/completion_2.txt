Operator plan:
1. Parse `DT_STRATA` into a datetime object to extract the month.
2. Iterate through hours 1 to 24. For each hour, calculate the mean of the corresponding `PCT_HOURLY` column, grouped by the month of the date.
3. Construct a result DataFrame with 24 rows (one per hour) and 12 columns (one per month, named january through december).
4. The `hour` column contains integers 1-24. The month columns contain the aggregated mean values.

```python
import pandas as pd
import numpy as np

def transform(source: pd.DataFrame) -> pd.DataFrame:
    # Parse dates to extract month
    dates = pd.to_datetime(source['DT_STRATA'], format='%m/%d/%Y')
    months = dates.dt.month
    
    # Initialize result dataframe
    result_data = []
    
    # Months mapping for column names
    month_names = [
        "january", "february", "march", "april", "may", "june",
        "july", "august", "september", "october", "november", "december"
    ]
    
    # Iterate through hours 1 to 24
    for h in range(1, 25):
        col_name = f'PCT_HOURLY_{h:02d}00'
        # Group by month and calculate mean
        # We need to align the hours with the months
        # Create a temporary series for this hour
        hourly_values = source[col_name]
        
        # Calculate mean for each month (1-12)
        # Using groupby on the months array
        grouped = pd.DataFrame({'month': months, 'value': hourly_values})
        means = grouped.groupby('month')['value'].mean()
        
        # Ensure we have values for all 12 months, fill NaN if any month is missing (though unlikely given data)
        row_data = [h] + [means.get(m, np.nan) for m in range(1, 13)]
        result_data.append(row_data)
        
    # Create final dataframe
    cols = ['hour'] + month_names
    result_df = pd.DataFrame(result_data, columns=cols)
    
    return result_df
```
