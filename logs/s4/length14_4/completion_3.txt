import pandas as pd
import numpy as np

def transform(source: pd.DataFrame) -> pd.DataFrame:
    # Parse Month from Time Period
    month_map = {
        'January': 1, 'February': 2, 'March': 3, 'April': 4,
        'May': 5, 'June': 6, 'July': 7, 'August': 8,
        'September': 9, 'October': 10, 'November': 11, 'December': 12
    }
    source['month'] = source['Time Period'].map(month_map)
    
    # Parse Number of Homes
    # Handle "0 (Modeled)" and other non-numeric strings
    homes_str = source['Number of Homes in Bin With This End Use Metered']
    # Extract digits from the string
    source['num_homes'] = homes_str.str.extract('(\d+)')[0].astype(float)
    
    # Calculate Total Demand for each row
    # Average Demand is per home. Total = Avg * Num Homes
    source['total_demand'] = source['Average Demand Per Home With This End Use [kW]'] * source['num_homes']
    
    # Initialize target columns
    target_cols = ['month', 'hour', 'hvac', 'domestic_water_heating', 'major_appliances', 'lighting', 'miscellaneous_plug_loads', 'total']
    
    # Group by Month and Hour Ending
    grouped = source.groupby(['month', 'Hour Ending'])
    
    results = []
    for (m, h), group in grouped:
        row = {'month': m, 'hour': h}
        
        # Initialize with 0
        row['hvac'] = 0.0
        row['domestic_water_heating'] = 0.0
        row['major_appliances'] = 0.0
        row['lighting'] = 0
        row['miscellaneous_plug_loads'] = 0.0
        row['total'] = 0
        
        # Map categories
        for _, r in group.iterrows():
            cat = r['End Use Category']
            end_use = r['End Use']
            demand = r['total_demand']
            
            if cat == 'Miscellaneous':
                row['miscellaneous_plug_loads'] += demand
            elif cat == 'Lighting':
                # Target lighting is 0, so we don't add it
                pass
            elif cat == 'Kitchen' or cat == 'Laundry':
                row['major_appliances'] += demand
            elif cat == 'Remaining Load':
                # Try to split Remaining Load into HVAC and DWH based on End Use string
                if 'Water' in end_use or 'Heater' in end_use:
                    row['domestic_water_heating'] += demand
                else:
                    row['hvac'] += demand
                    
        results.append(row)
        
    df_target = pd.DataFrame(results)
    
    # Ensure correct dtypes
    df_target['month'] = df_target['month'].astype(int)
    df_target['hour'] = df_target['hour'].astype(int)
    df_target['light
