Operator plan: Parse the `Date` column (format `%a %m/%d/%Y`); format `dt_strata` as `%m/%d/%y`; compute `dow` as ISO weekday (Monday=1..Sunday=7); rename the 24 hourly columns to `pct_hourly_HH00`; add `pct_hourly_2500` as NaN.

```python
import pandas as pd
import numpy as np

def transform(source: pd.DataFrame) -> pd.DataFrame:
    dates = pd.to_datetime(source['Date'], format='%a %m/%d/%Y')
    out = pd.DataFrame(index=source.index)
    out['dt_strata'] = dates.dt.strftime('%m/%d/%y')
    out['dow'] = dates.dt.dayofweek + 1
    
    for h in range(1, 25):
        src_col = f'{h}:00' if h < 24 else '24:00:00'
        out[f'pct_hourly_{h:02d}00'] = source[src_col]
    
    out['pct_hourly_2500'] = np.nan
    
    return out.reset_index(drop=True)
```
