import pandas as pd
import numpy as np

def transform(source: pd.DataFrame) -> pd.DataFrame:
    # Parse time
    dt = pd.to_datetime(source['time'])
    hour = dt.dt.hour
    month = dt.dt.month
    
    # Define seasons
    # Winter: 12, 1, 2
    # Spring: 3, 4, 5
    # Summer: 6, 7, 8
    # Fall: 9, 10, 11
    def get_season(m):
        if m in [12, 1, 2]:
            return 'Winter'
        elif m in [3, 4, 5]:
            return 'Spring'
        elif m in [6, 7, 8]:
            return 'Summer'
        else:
            return 'Fall'
            
    season = month.apply(get_season)
    
    # Define component sums
    hvac_cols = [
        'electricity_heating_kwh',
        'electricity_central_system_heating_kwh',
        'electricity_cooling_kwh',
        'electricity_central_system_cooling_kwh',
        'electricity_fans_heating_kwh',
        'electricity_fans_cooling_kwh',
        'electricity_pumps_heating_kwh',
        'electricity_central_system_pumps_heating_kwh',
        'electricity_central_system_pumps_cooling_kwh',
        'electricity_pumps_cooling_kwh'
    ]
    
    dwh_col = 'electricity_water_systems_kwh'
    
    major_appliances_cols = [
        'electricity_refrigerator_kwh',
        'electricity_clothes_washer_kwh',
        'electricity_clothes_dryer_kwh',
        'electricity_cooking_range_kwh',
        'electricity_dishwasher_kwh'
    ]
    
    lighting_cols = [
        'electricity_interior_lighting_kwh',
        'electricity_exterior_lighting_kwh'
    ]
    
    misc_plug_cols = [
        'electricity_plug_loads_kwh',
        'electricity_house_fan_kwh',
        'electricity_range_fan_kwh',
        'electricity_bath_fan_kwh',
        'electricity_ceiling_fan_kwh',
        'electricity_pool_heater_kwh',
        'electricity_pool_pump_kwh',
        'electricity_hot_tub_heater_kwh',
        'electricity_hot_tub_pump_kwh'
    ]
    
    total_col = 'net_site_electricity_kwh'
    
    # Create a working dataframe
    df = pd.DataFrame()
    df['season'] = season
    df['hour'] = hour
    
    df['hvac'] = source[hvac_cols].sum(axis=1)
    df['domestic_water_heating'] = source[dwh_col]
    df['major_appliances'] = source[major_appliances_cols].sum(axis=1)
    df['lighting'] = source[lighting_cols].sum(axis=1)
    df['miscellaneous_plug_loads'] = source[misc_plug_cols].sum(axis=1)
    df['total'] = source[total_col]
    
    # Group by season and hour, sum the values
    result = df.groupby(['season', 'hour'], as_index=False).sum()
    
    # Ensure column order
    result = result[['season', 'hour', 'hvac', 'domestic_water_heating', 'major_appliances', 'lighting', 'miscellaneous_plug_loads', 'total']]
    
    return result
