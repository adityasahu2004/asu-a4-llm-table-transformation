def transform(source: pd.DataFrame) -> pd.DataFrame:

    # 12,1,2 -> Summer
    
    def get_season(month):
        if month in [6, 7, 8]:
            return "Winter"
        elif month in [9, 10, 11]:
            return "Spring"
        elif month in [12, 1, 2]:
            return "Summer"
        else:

            return "Autumn" 

    df['season'] = df['month'].apply(get_season)
