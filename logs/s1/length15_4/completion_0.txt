import pandas as pd
import numpy as np

def transform(source: pd.DataFrame) -> pd.DataFrame:
    # Create a copy to avoid modifying the original
    df = source.copy()
    
    # Map Time Period to Season
    def get_season(time_period):
        if pd.isna(time_period):
            return None
        tp = str(time_period).strip()
        if tp in ['December', 'January', 'February']:
            return 'Winter'
        elif tp in ['March', 'April', 'May']:
            return 'Spring'
        elif tp in ['June', 'July', 'August']:
            return 'Summer'
        elif tp in ['September', 'October', 'November']:
            return 'Fall'
        else:
            # Non-month values map to null season
            return None
            
    df['season'] = df['Time Period'].apply(get_season)
    
    # Map End Use Category and End Use to target columns
    # We assume the following mapping based on typical energy end-use categories:
    # major_appliances: Kitchen, Laundry
    # miscellaneous_plug_loads: Miscellaneous
    # hvac: Remaining Load with End Use 'HVAC' (or similar)
    # domestic_water_heating: Remaining Load with End Use 'Water Heater' (or similar)
    # lighting: Lighting (but target is 0, so we might exclude or set to 0)
    
    # Initialize target columns with 0
    df['hvac'] = 0.0
    df['domestic_water_heating'] = 0.0
    df['major_appliances'] = 0.0
    df['miscellaneous_plug_loads'] = 0.0
    
    # Map major_appliances
    mask_major = df['End Use Category'].isin(['Kitchen', 'Laundry'])
    df.loc[mask_major, 'major_appliances'] = df.loc[mask_major, 'Average Demand Per Home With This End Use [kW]']
    
    # Map miscellaneous_plug_loads
    mask_misc = df['End Use Category'] == 'Miscellaneous'
    df.loc[mask_misc, 'miscellaneous_plug_loads'] = df.loc[mask_misc, 'Average Demand Per Home With This End Use [kW]']
    
    # Map hvac and domestic_water_heating from Remaining Load
    # We need to distinguish between HVAC and Water Heater within Remaining Load
    # Assuming End Use strings are 'HVAC' and 'Water Heater'
    mask_remaining = df['End Use Category'] == 'Remaining Load'
    
    # Check for HVAC
    mask_hvac = mask_remaining & (df['End Use'] == 'HVAC')
    df.loc[mask_hvac, 'hvac'] = df.loc[mask_hvac, 'Average Demand Per Home With This End Use [kW]']
    
    # Check for Water Heater
    mask_wh = mask_remaining & (df['End Use'] == 'Water Heater')
    df.loc[mask_wh, 'domestic_water_heating'] = df.loc[mask_wh, 'Average Demand Per Home With This End Use [kW]']
    
    # If there are other End Uses in Remaining Load, they are ignored for now
    # If the mapping is incorrect, the values will be 0.
    
    # Group by season and hour
    # Note: season can be null
    grouped = df.groupby(['season', 'Hour
